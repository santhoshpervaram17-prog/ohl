API v3 Ready
Enter your Fyers API v3 credentials to query GET https://api-t1.fyers.in/data/quotes in batches of 50 symbols. Credentials stay strictly in your browser memory / local storage.

Header sent: Authorization: <app_id>:<access_token>

If your browser blocks direct cross-origin requests (CORS) to api-t1.fyers.in, run this lightweight Python script with pip install fyers-apiv3. It waits until 09:16:00 IST, fetches all Large & Mid Cap quotes in one batch, and prints/serves the Open=High and Open=Low lists: 

from fyers_apiv3 import fyersModel
import time, datetime

APP_ID = "YOUR_APP_ID-100"       # Replace with your Fyers App ID
ACCESS_TOKEN = "YOUR_TOKEN"      # Replace with your daily Access Token

# Large & Mid Cap NSE Universe (up to 50 symbols per Quotes API call)
SYMBOLS = [
    "NSE:RELIANCE-EQ", "NSE:TCS-EQ", "NSE:HDFCBANK-EQ", "NSE:ICICIBANK-EQ",
    "NSE:INFY-EQ", "NSE:SBIN-EQ", "NSE:BHARTIARTL-EQ", "NSE:LT-EQ",
    "NSE:TATAMOTORS-EQ", "NSE:AXISBANK-EQ", "NSE:SUNPHARMA-EQ", "NSE:NTPC-EQ",
    "NSE:BAJFINANCE-EQ", "NSE:M&M-EQ", "NSE:BEL-EQ", "NSE:TRENT-EQ",
    "NSE:PERSISTENT-EQ", "NSE:COFORGE-EQ", "NSE:DIXON-EQ", "NSE:POLYCAB-EQ",
    "NSE:FEDERALBNK-EQ", "NSE:INDHOTEL-EQ", "NSE:CUMMINSIND-EQ", "NSE:LUPIN-EQ",
    "NSE:VOLTAS-EQ", "NSE:MAXHEALTH-EQ", "NSE:OBEROIRLTY-EQ", "NSE:ASHOKLEY-EQ"
]

fyers = fyersModel.FyersModel(client_id=APP_ID, is_async=False, token=ACCESS_TOKEN, log_path="")

def scan_ohl():
    resp = fyers.quotes(data={"symbols": ",".join(SYMBOLS)})
    if resp.get("s") != "ok":
        print("Error:", resp)
        return
    open_high, open_low = [], []
    for item in resp.get("d", []):
        v = item.get("v", {})
        sym = item.get("n", v.get("symbol", ""))
        o, h, l, ltp, chp = v.get("open_price"), v.get("high_price"), v.get("low_price"), v.get("lp"), v.get("chp")
        if o and h and o == h:
            open_high.append((sym, o, l, ltp, chp))
        if o and l and o == l:
            open_low.append((sym, o, h, ltp, chp))

    print(f"\n=== OPEN = HIGH ({len(open_high)}) ===")
    for s, o, l, ltp, chp in open_high:
        print(f"{s:20s} | Open=High: {o:8.2f} | Low: {l:8.2f} | LTP: {ltp:8.2f} ({chp:+.2f}%)")

    print(f"\n=== OPEN = LOW ({len(open_low)}) ===")
    for s, o, h, ltp, chp in open_low:
        print(f"{s:20s} | Open=Low:  {o:8.2f} | High:{h:8.2f} | LTP: {ltp:8.2f} ({chp:+.2f}%)")

if __name__ == "__main__":
    scan_ohl()

Option 2 (Python Server): Run app.py on GitHub Codespaces, PythonAnywhere, or Render to serve /api/quotes dynamically.

Step 1: Enable Fyers TOTP

Go to myaccount.fyers.in → Manage Account → External 2FA TOTP. Copy the secret 32-character TOTP Key.

Step 2: Create API App

In myapi.fyers.in/dashboard, create an app and copy your App ID (-100) and Secret ID.

Step 3: Deploy Free Server

Push the provided fyers_ohl_cloud_server.zip to GitHub & connect to Render.com with your env variables.

# app.py — Free Cloud Server with Daily Auto-Authentication (TOTP + Token Cache)
import os, json, base64, datetime, time
from urllib.parse import urlparse, parse_qs
import pyotp, requests
from flask import Flask, jsonify, send_file, request
from fyers_apiv3 import fyersModel

app = Flask(__name__)
TOKEN_FILE = "fyers_daily_token.json"

def get_auto_token():
    today = datetime.date.today().strftime("%Y-%m-%d")
    if os.path.exists(TOKEN_FILE):
        data = json.load(open(TOKEN_FILE))
        if data.get("date") == today and data.get("access_token"):
            return data["access_token"]

    b64 = lambda s: base64.b64encode(str(s).encode("ascii")).decode("ascii")
    fy_id = os.environ["FYERS_USER_ID"]
    app_id = os.environ["FYERS_APP_ID"]        # e.g. XXXXXXXX-100
    secret_id = os.environ["FYERS_SECRET_ID"]
    totp_key = os.environ["FYERS_TOTP_KEY"]
    pin = os.environ["FYERS_PIN"]
    redirect_uri = os.environ.get("FYERS_REDIRECT_URI", "https://127.0.0.1:5000/")

    # 1. Send Login OTP
    r1 = requests.post("https://api-t2.fyers.in/vagator/v2/send_login_otp_v2",
                       json={"fy_id": b64(fy_id), "app_id": "2"}).json()
    if datetime.datetime.now().second % 30 > 27:
        time.sleep(4)
    # 2. Verify TOTP
    otp = pyotp.TOTP(totp_key).now()
    r2 = requests.post("https://api-t2.fyers.in/vagator/v2/verify_otp",
                       json={"request_key": r1["request_key"], "otp": otp}).json()
    # 3. Verify PIN
    ses = requests.Session()
    r3 = ses.post("https://api-t2.fyers.in/vagator/v2/verify_pin_v2",
                  json={"request_key": r2["request_key"], "identity_type": "pin", "identifier": b64(pin)}).json()
    ses.headers.update({"authorization": f"Bearer {r3['data']['access_token']}"})
    # 4. Exchange for v3 Auth Code & Access Token
    r4 = ses.post("https://api-t1.fyers.in/api/v3/token", json={
        "fyers_id": fy_id, "app_id": app_id[:-4], "redirect_uri": redirect_uri,
        "appType": "100", "code_challenge": "", "state": "auto", "scope": "",
        "nonce": "", "response_type": "code", "create_cookie": True
    }).json()
    auth_code = parse_qs(urlparse(r4["Url"]).query)["auth_code"][0]
    session = fyersModel.SessionModel(client_id=app_id, secret_key=secret_id,
                                      redirect_uri=redirect_uri, response_type="code",
                                      grant_type="authorization_code")
    session.set_token(auth_code)
    token = session.generate_token()["access_token"]
    json.dump({"date": today, "access_token": token}, open(TOKEN_FILE, "w"))
    return token

@app.route("/")
def index():
    return send_file("fyers_ohl_scanner.html")

@app.route("/api/quotes")
def api_quotes():
    symbols = request.args.get("symbols", "")
    token = get_auto_token()
    fyers = fyersModel.FyersModel(client_id=os.environ["FYERS_APP_ID"], is_async=False, token=token, log_path="")
    return jsonify(fyers.quotes(data={"symbols": symbols}))
